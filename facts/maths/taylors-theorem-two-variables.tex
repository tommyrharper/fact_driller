% Taylor's theorem for functions of more than one variable
\[
\begin{aligned}
f(x + \delta x, y + \delta y) = {} & f(x, y) + \delta x \frac{\partial f}{\partial x} + \delta y \frac{\partial f}{\partial y} \\
& + \frac{1}{2!} \left( (\delta x)^2 \frac{\partial^2 f}{\partial x^2} + 2 \delta x \delta y \frac{\partial^2 f}{\partial x \partial y} + (\delta y)^2 \frac{\partial^2 f}{\partial y^2} \right) + \cdots
\end{aligned}
\]

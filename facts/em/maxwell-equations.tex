% Maxwell's equations
\[
\begin{aligned}
\boldsymbol{\nabla} \cdot \boldsymbol{D} &= \rho_{\text{free}} \\
\boldsymbol{\nabla} \times \boldsymbol{E} &= -\frac{\partial \boldsymbol{B}}{\partial t} \\
\boldsymbol{\nabla} \cdot \boldsymbol{B} &= 0 \\
\boldsymbol{\nabla} \times \boldsymbol{H} &= \boldsymbol{J}_{\text{free}} + \frac{\partial \boldsymbol{D}}{\partial t}
\end{aligned}
\]
